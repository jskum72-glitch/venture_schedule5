import React, { useState } from 'react';
import { Calendar, Clock, CheckCircle2, Send } from 'lucide-react';

const VentureScheduleApp = () => {
  const [step, setStep] = useState(1); // 1: 위원님 설정, 2: 거래처 확인, 3: 완료
  const [selectedDate, setSelectedDate] = useState('');
  const [selectedTime, setSelectedTime] = useState('');
  
  // 기본 메시지 정보
  const evaluatorName = "금정섭 위원";
  const organization = "한국 기업 데이터";
  const purpose = "벤처기업 현장 실사 일정 조율";

  const availableTimes = ["10:00", "14:00", "16:00"];

  // 거래처에 보낼 최종 메시지 서식
  const finalMessage = `안녕하세요, ${organization}의 ${evaluatorName}입니다. 
${purpose}을 위해 아래 일정 중 가능한 시간을 선택해 주세요.`;

  const handleComplete = () => {
    setStep(3);
  };

  return (
    <div className="max-w-md mx-auto min-h-screen bg-gray-50 p-6 font-sans">
      {/* Header */}
      <div className="mb-8 text-center">
        <h1 className="text-xl font-bold text-blue-600">Venture-Sync</h1>
        <p className="text-sm text-gray-500">벤처인증 현장평가 일정 조율</p>
      </div>

      {step === 1 && (
        <div className="bg-white rounded-2xl p-6 shadow-sm border border-gray-100">
          <h2 className="text-lg font-semibold mb-4 text-gray-800">실사 가능 일정 제안</h2>
          <div className="space-y-4">
            <div>
              <label className="block text-sm font-medium text-gray-600 mb-1">방문 예정 날짜</label>
              <input 
                type="date" 
                className="w-full p-3 border rounded-xl focus:ring-2 focus:ring-blue-500 outline-none"
                onChange={(e) => setSelectedDate(e.target.value)}
              />
            </div>
            <div className="p-4 bg-blue-50 rounded-xl">
              <p className="text-sm text-blue-700 leading-relaxed whitespace-pre-line">
                <strong>[발송 메시지 미리보기]</strong><br/>
                {finalMessage}
              </p>
            </div>
            <button 
              onClick={() => setStep(2)}
              className="w-full bg-blue-600 text-white py-4 rounded-xl font-bold flex justify-center items-center gap-2"
            >
              <Send size={18} /> 일정 조율 링크 생성
            </button>
          </div>
        </div>
      )}

      {step === 2 && (
        <div className="bg-white rounded-2xl p-6 shadow-lg border border-blue-100">
          <div className="mb-6">
            <span className="bg-blue-100 text-blue-700 text-xs font-bold px-3 py-1 rounded-full text-center">수신자 화면</span>
            <h2 className="text-lg font-bold mt-3 text-gray-800">{organization}</h2>
            <p className="text-gray-600 mt-1">{evaluatorName}님이 일정을 요청했습니다.</p>
          </div>

          <div className="space-y-4">
            <div className="flex items-center gap-3 p-3 border-b">
              <Calendar className="text-blue-500" size={20} />
              <span className="font-medium">{selectedDate || '2023-11-20'} (예정)</span>
            </div>
            
            <p className="text-sm font-medium text-gray-500 mt-4">방문 가능한 시간을 선택해 주세요:</p>
            <div className="grid grid-cols-1 gap-2">
              {availableTimes.map((time) => (
                <button
                  key={time}
                  onClick={() => setSelectedTime(time)}
                  className={`py-3 px-4 rounded-xl border text-left transition-all ${
                    selectedTime === time 
                    ? 'border-blue-500 bg-blue-50 text-blue-700 font-bold' 
                    : 'border-gray-200 hover:bg-gray-50'
                  }`}
                >
                  {time}
                </button>
              ))}
            </div>

            <button 
              onClick={handleComplete}
              disabled={!selectedTime}
              className={`w-full py-4 rounded-xl font-bold mt-4 ${
                selectedTime ? 'bg-gray-900 text-white' : 'bg-gray-200 text-gray-400'
              }`}
            >
              선택한 일정으로 확정하기
            </button>
          </div>
        </div>
      )}

      {step === 3 && (
        <div className="text-center bg-white rounded-2xl p-10 shadow-sm border">
          <CheckCircle2 size={60} className="mx-auto text-green-500 mb-4" />
          <h2 className="text-xl font-bold text-gray-800">일정 확정 완료</h2>
          <p className="text-gray-500 mt-2">
            {selectedDate} {selectedTime}에<br/>
            현장 실사가 예약되었습니다.
          </p>
          <div className="mt-6 p-4 bg-gray-50 rounded-lg text-sm text-left">
            <p>• <strong>담당자:</strong> {evaluatorName}</p>
            <p>• <strong>장소:</strong> 거래처 본사</p>
          </div>
          <button 
            onClick={() => setStep(1)}
            className="mt-8 text-blue-600 font-medium text-sm underline"
          >
            처음으로 돌아가기
          </button>
        </div>
      )}
    </div>
  );
};

export default VentureScheduleApp;
